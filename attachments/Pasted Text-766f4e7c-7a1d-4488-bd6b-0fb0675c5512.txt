/*!
 * Sour Paint Studios — Audio Engine (pitch + playing-feel detection)
 * ---------------------------------------------------------------------------
 * Plain browser JavaScript, no dependencies, no build step. Works in mobile
 * Safari/Chrome. Proven inside the Sour Paint Songbook tuner and guitar
 * painter.
 *
 * WHAT IT GIVES YOU (every animation frame, via your onFrame callback):
 *   frame = {
 *     rms        raw loudness of the last ~11 ms (0..~1)
 *     level      loudness relative to the loudest recent playing (0..1)
 *     sounding   true when something is actually playing (noise-gated)
 *     onset      true on the frame a pick attack / new note starts
 *     strength   attack hardness 0..1 (only meaningful when onset is true)
 *     rate       attacks per second over the last 2.5 s (smoothed)
 *     speed      playing speed 0..1 (0 = slow/sparse, 1 = fast/frantic)
 *     freq       detected pitch in Hz, or 0 if none this frame
 *     midi       pitch as a (fractional) MIDI note number, or null
 *     pitchNorm  last known pitch mapped 0..1 across E2 (low) .. E6 (high)
 *     note       note name like "A", "C#" (nearest note) or ""
 *     octave     octave number (A4 = 440 Hz) or null
 *     cents      how many cents sharp(+)/flat(-) of the nearest note
 *     clarity    pitch confidence 0..1 (>= 0.88 is a clean tuner reading)
 *   }
 *
 * USAGE:
 *   var listener = new SourPaintListener({ a4: 440 });
 *   startButton.onclick = function () {          // MUST be inside a tap/click
 *     listener.start(function (frame) {
 *       // drive your canvas / art from `frame` here
 *     }).catch(function (err) { console.log(err.name, err.message); });
 *   };
 *   // listener.stop();  when done
 *
 * Needs HTTPS (or localhost) and the user's microphone permission.
 *
 * HOW THE PITCH DETECTOR WORKS (McLeod Pitch Method / "MPM"):
 *   A ~40 ms window of audio is auto-correlated and normalised (NSDF). The
 *   first strong peak is chosen (within 90% of the best peak), then refined
 *   to sub-sample accuracy with parabolic interpolation. This avoids the
 *   classic octave errors of plain autocorrelation on guitar. Verified to
 *   within ~0.6 cent on synthetic guitar-like tones.
 * ---------------------------------------------------------------------------
 */
(function (root) {
  'use strict';

  var SHARPS = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B'];

  // ===== 1. Pitch detection core (McLeod Pitch Method) =====================
  // Runs on the first W samples of buf (a Float32Array of time-domain audio).
  // Lags outside [minLag, maxLag] are ignored; they bound the detectable range.
  // Returns { freq, clarity } (clarity 0..1), or null if nothing clear.
  var mpmScratch = null;
  function detectPitchMPM(buf, W, sr, minLag, maxLag) {
    if (maxLag > W - 2) maxLag = W - 2;
    if (!mpmScratch || mpmScratch.length < maxLag + 1) mpmScratch = new Float32Array(maxLag + 1);
    var nsdf = mpmScratch;
    var m = 0, j, tau;
    for (j = 0; j < W; j++) m += buf[j] * buf[j];
    m *= 2;
    if (m <= 0) return null;
    for (tau = 0; tau <= maxLag; tau++) {
      if (tau > 0) m -= buf[tau - 1] * buf[tau - 1] + buf[W - tau] * buf[W - tau];
      var r = 0;
      for (var k = 0, n = W - tau; k < n; k++) r += buf[k] * buf[k + tau];
      nsdf[tau] = m > 1e-12 ? (2 * r) / m : 0;
    }
    // Key maxima: the highest point of each positive lobe, after the
    // zero-lag lobe has dropped below zero.
    var peaks = [];
    tau = 1;
    while (tau < maxLag && nsdf[tau] > 0) tau++;
    var inPos = false, best = 0, bestTau = -1;
    for (; tau < maxLag; tau++) {
      if (!inPos) {
        if (nsdf[tau] > 0 && nsdf[tau - 1] <= 0) { inPos = true; best = nsdf[tau]; bestTau = tau; }
      } else if (nsdf[tau] <= 0) {
        inPos = false;
        peaks.push(bestTau);
      } else if (nsdf[tau] > best) {
        best = nsdf[tau]; bestTau = tau;
      }
    }
    if (inPos && bestTau > 0 && bestTau < maxLag - 1) peaks.push(bestTau);
    var highest = 0, i;
    for (i = 0; i < peaks.length; i++) {
      if (peaks[i] >= minLag && nsdf[peaks[i]] > highest) highest = nsdf[peaks[i]];
    }
    if (highest <= 0) return null;
    var chosen = -1;
    for (i = 0; i < peaks.length; i++) {
      if (peaks[i] >= minLag && nsdf[peaks[i]] >= 0.9 * highest) { chosen = peaks[i]; break; }
    }
    if (chosen < 1) return null;
    var a = nsdf[chosen - 1], b = nsdf[chosen], c = nsdf[chosen + 1];
    var denom = a - 2 * b + c;
    var shift = denom !== 0 ? 0.5 * (a - c) / denom : 0;
    if (shift > 1 || shift < -1) shift = 0;
    return { freq: sr / (chosen + shift), clarity: b - 0.25 * (a - c) * shift };
  }

  // ===== 2. Helpers ========================================================
  function freqToMidi(f, a4) { return 69 + 12 * Math.log(f / (a4 || 440)) / Math.LN2; }
  function midiToFreq(m, a4) { return (a4 || 440) * Math.pow(2, (m - 69) / 12); }
  function median(arr) {
    var s = arr.slice().sort(function (x, y) { return x - y; });
    return s[Math.floor(s.length / 2)];
  }

  // ===== 3. Live microphone listener =======================================
  function SourPaintListener(opts) {
    opts = opts || {};
    this.a4 = opts.a4 || 440;
    this.minClarity = opts.minClarity || 0.88;   // tuner-grade pitch confidence
    this.onFrameCb = null;
    this.running = false;
    this._reset();
  }

  SourPaintListener.prototype._reset = function () {
    this.ctx = null; this.stream = null; this.source = null; this.analyser = null;
    this.buf = null; this.W = 0; this.minLag = 0; this.maxLag = 0; this.raf = 0;
    this.hist = [];            // recent pitch readings (median-smoothed)
    this.rhist = [];           // recent loudness readings (for onset detection)
    this.onsets = [];          // timestamps of recent attacks
    this.peak = 0.1; this.prevR = 0; this.rate = 0;
    this.lastOnset = 0; this.lastPitchTs = 0; this.lastHeard = 0;
    this.pitchNorm = 0.5; this.midi = null; this.freq = 0; this.clarity = 0;
  };

  // MUST be called from inside a user tap/click (phones require that to start
  // audio). Returns a Promise; rejects with the getUserMedia error if the mic
  // is denied/unavailable (err.name: NotAllowedError, NotFoundError, ...).
  SourPaintListener.prototype.start = function (onFrame) {
    var self = this;
    if (self.running) return Promise.resolve();
    self.onFrameCb = onFrame;
    var Ctx = window.AudioContext || window.webkitAudioContext;
    if (!Ctx || !navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
      return Promise.reject(new Error('Microphone not supported on this page (needs HTTPS).'));
    }
    self._reset();
    self.ctx = new Ctx();
    if (self.ctx.state === 'suspended') self.ctx.resume();
    return navigator.mediaDevices.getUserMedia({
      // Turn OFF the phone's voice processing: it mangles instruments.
      audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false },
      video: false
    }).then(function (stream) {
      var ctx = self.ctx;
      self.stream = stream;
      var src = ctx.createMediaStreamSource(stream);
      // Trim rumble below the low E and hiss above the useful range.
      var hp = ctx.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 50;
      var lp = ctx.createBiquadFilter(); lp.type = 'lowpass'; lp.frequency.value = 1800;
      var an = ctx.createAnalyser();
      var W = 2048;
      while (W < ctx.sampleRate * 0.04) W *= 2;     // ~40 ms analysis window
      an.fftSize = W;
      an.smoothingTimeConstant = 0;
      src.connect(hp); hp.connect(lp); lp.connect(an);
      self.source = src; self.analyser = an; self.W = W;
      self.buf = new Float32Array(W);
      self.minLag = Math.floor(ctx.sampleRate / 1400);               // top pitch ~1400 Hz
      self.maxLag = Math.min(W - 2, Math.ceil(ctx.sampleRate / 58)); // bottom ~58 Hz
      self.running = true;
      var loop = function (ts) {
        if (!self.running) return;
        self.raf = requestAnimationFrame(loop);
        self._tick(ts);
      };
      self.raf = requestAnimationFrame(loop);
    });
  };

  SourPaintListener.prototype.stop = function () {
    this.running = false;
    if (this.raf) cancelAnimationFrame(this.raf);
    if (this.source) { try { this.source.disconnect(); } catch (e) {} }
    if (this.stream) this.stream.getTracks().forEach(function (t) { t.stop(); });
    if (this.ctx) { try { this.ctx.close(); } catch (e) {} }
    this._reset();
  };

  SourPaintListener.prototype._tick = function (ts) {
    var buf = this.buf, n = buf.length, i;
    this.analyser.getFloatTimeDomainData(buf);

    // --- loudness (last ~11 ms) ---
    var tail = Math.min(512, n), sum = 0;
    for (i = n - tail; i < n; i++) sum += buf[i] * buf[i];
    var rms = Math.sqrt(sum / tail);
    this.peak = Math.max(rms, this.peak * 0.9993, 0.03);   // slowly-decaying loudest
    var level = Math.min(1, rms / this.peak);
    var sounding = rms > 0.006;                            // noise gate
    var now = Date.now();

    // --- onset (pick attack): sharp rise vs. loudness ~50 ms ago ---
    var rh = this.rhist; rh.push(rms); if (rh.length > 6) rh.shift();
    var ref = rh.length >= 5 ? Math.min(rh[rh.length - 4], rh[rh.length - 3], rh[rh.length - 2]) : rms;
    var onset = sounding &&
      rms - ref > Math.max(0.008, 0.1 * this.peak) &&
      rms > ref * 1.25 && rms > this.prevR &&
      now - this.lastOnset > 70;                           // 70 ms refractory
    this.prevR = rms;
    var strength = 0;
    if (onset) {
      this.lastOnset = now; this.onsets.push(now);
      strength = Math.max(0.15, Math.min(1, level * 0.8 + (rms - ref) * 2));
    }

    // --- playing speed: attacks per second over the last 2.5 s ---
    var cutoff = now - 2500;
    while (this.onsets.length && this.onsets[0] < cutoff) this.onsets.shift();
    this.rate += (this.onsets.length / 2.5 - this.rate) * 0.05;
    var speed = Math.max(0, Math.min(1, (this.rate - 0.8) / 5.5));

    // --- pitch (every ~30 ms; skipped when silent) ---
    var freq = 0;
    if (ts - this.lastPitchTs >= 30) {
      this.lastPitchTs = ts;
      if (sounding) {
        var p = detectPitchMPM(buf, this.W, this.ctx.sampleRate, this.minLag, this.maxLag);
        if (p && p.clarity >= 0.8 && p.freq >= 60 && p.freq <= 1400) {
          this.clarity = p.clarity;
          var m = freqToMidi(p.freq, this.a4);
          // pitchNorm accepts looser clarity (good for art); the tuner-grade
          // reading (freq/midi/note/cents) needs clarity >= minClarity.
          this.pitchNorm = Math.max(0, Math.min(1, (m - 40) / 48));
          if (p.clarity >= this.minClarity) {
            var h = this.hist;
            // A jump over ~half a semitone is a new note: restart smoothing.
            if (h.length && Math.abs(m - median(h)) > 0.6) h.length = 0;
            h.push(m); if (h.length > 7) h.shift();
            this.lastHeard = now;
            if (h.length >= 2) { this.midi = median(h); this.freq = midiToFreq(this.midi, this.a4); }
          }
        }
      }
      if (this.lastHeard && now - this.lastHeard > 1500) {   // note has ended
        this.lastHeard = 0; this.hist.length = 0; this.midi = null; this.freq = 0;
      }
    }

    var midi = this.midi, name = '', oct = null, cents = 0;
    if (midi !== null) {
      var nearest = Math.round(midi);
      name = SHARPS[((nearest % 12) + 12) % 12];
      oct = Math.floor(nearest / 12) - 1;
      cents = (midi - nearest) * 100;
    }
    if (this.onFrameCb) {
      this.onFrameCb({
        rms: rms, level: level, sounding: sounding, onset: onset, strength: strength,
        rate: this.rate, speed: speed,
        freq: this.freq, midi: midi, pitchNorm: this.pitchNorm,
        note: name, octave: oct, cents: cents, clarity: this.clarity
      });
    }
  };

  root.SourPaintListener = SourPaintListener;
  root.SourPaintAudio = { detectPitchMPM: detectPitchMPM, freqToMidi: freqToMidi, midiToFreq: midiToFreq };
})(typeof window !== 'undefined' ? window : this);
